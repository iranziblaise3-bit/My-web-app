/* =========================================================================
   WEB AUDIO API SYNTHESIZER & HAPTICS
   ========================================================================= */

class SoundEngine {
  private ctx: AudioContext | null = null;
  private isMuted: boolean = false;

  private getContext(): AudioContext | null {
    if (typeof window === 'undefined') return null;
    if (!this.ctx) {
      const AudioCtx =
        window.AudioContext ||
        (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
      if (AudioCtx) {
        this.ctx = new AudioCtx();
      }
    }
    if (this.ctx && this.ctx.state === 'suspended') {
      this.ctx.resume().catch(() => {});
    }
    return this.ctx;
  }

  public init() {
    this.getContext();
  }

  public setMuted(muted: boolean) {
    this.isMuted = muted;
  }

  /**
   * Ultra-sharp, crisp mechanical tournament clock switch sound.
   */
  public playMechanicalClick() {
    if (this.isMuted) return;
    const ctx = this.getContext();
    if (!ctx) return;

    try {
      const now = ctx.currentTime;

      // 1. Ultra-sharp transient strike (high-frequency downward snap)
      const sharpOsc = ctx.createOscillator();
      const sharpGain = ctx.createGain();
      sharpOsc.type = 'triangle';
      sharpOsc.frequency.setValueAtTime(3600, now);
      sharpOsc.frequency.exponentialRampToValueAtTime(320, now + 0.015);

      sharpGain.gain.setValueAtTime(1.0, now);
      sharpGain.gain.exponentialRampToValueAtTime(0.001, now + 0.015);

      sharpOsc.connect(sharpGain);
      sharpGain.connect(ctx.destination);

      sharpOsc.start(now);
      sharpOsc.stop(now + 0.018);

      // 2. Crisp metallic/plastic snap (high-pass filtered white noise burst)
      const bufferSize = Math.floor(ctx.sampleRate * 0.018);
      const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
      const data = buffer.getChannelData(0);
      for (let i = 0; i < bufferSize; i++) {
        data[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / bufferSize, 2);
      }

      const noise = ctx.createBufferSource();
      noise.buffer = buffer;

      const filter = ctx.createBiquadFilter();
      filter.type = 'highpass';
      filter.frequency.setValueAtTime(3200, now);
      filter.Q.setValueAtTime(2.5, now);

      const noiseGain = ctx.createGain();
      noiseGain.gain.setValueAtTime(0.95, now);
      noiseGain.gain.exponentialRampToValueAtTime(0.001, now + 0.016);

      noise.connect(filter);
      filter.connect(noiseGain);
      noiseGain.connect(ctx.destination);

      noise.start(now);
      noise.stop(now + 0.018);

      // 3. Crisp secondary microswitch mechanical snap (at +4ms)
      const snapOsc = ctx.createOscillator();
      const snapGain = ctx.createGain();
      snapOsc.type = 'square';
      snapOsc.frequency.setValueAtTime(2200, now + 0.004);
      snapOsc.frequency.exponentialRampToValueAtTime(450, now + 0.02);

      const snapFilter = ctx.createBiquadFilter();
      snapFilter.type = 'bandpass';
      snapFilter.frequency.setValueAtTime(1800, now + 0.004);

      snapGain.gain.setValueAtTime(0.001, now);
      snapGain.gain.setValueAtTime(0.7, now + 0.004);
      snapGain.gain.exponentialRampToValueAtTime(0.001, now + 0.02);

      snapOsc.connect(snapFilter);
      snapFilter.connect(snapGain);
      snapGain.connect(ctx.destination);

      snapOsc.start(now + 0.004);
      snapOsc.stop(now + 0.022);
    } catch {
      // Audio failsafe
    }
  }

  /**
   * Urgent low-time warning beep (fires once per second when remaining time <= 10s).
   */
  public playLowTimeWarning(remainingSeconds: number) {
    if (this.isMuted) return;
    const ctx = this.getContext();
    if (!ctx) return;

    try {
      const now = ctx.currentTime;
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();

      const baseFreq = remainingSeconds <= 3 ? 1046.5 : remainingSeconds <= 5 ? 987.77 : 880.0;
      osc.type = 'square';
      osc.frequency.setValueAtTime(baseFreq, now);

      const filter = ctx.createBiquadFilter();
      filter.type = 'lowpass';
      filter.frequency.setValueAtTime(2400, now);

      gain.gain.setValueAtTime(0.001, now);
      gain.gain.linearRampToValueAtTime(0.45, now + 0.008);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);

      osc.connect(filter);
      filter.connect(gain);
      gain.connect(ctx.destination);

      osc.start(now);
      osc.stop(now + 0.1);
    } catch {
      // Audio failsafe
    }
  }

  /**
   * Loud timeout buzzer / alarm when flag falls (0:00).
   */
  public playTimeoutAlarm() {
    if (this.isMuted) return;
    const ctx = this.getContext();
    if (!ctx) return;

    try {
      const now = ctx.currentTime;
      const freqs = [620, 520, 415];
      freqs.forEach((freq, idx) => {
        const startTime = now + idx * 0.14;
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();

        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(freq, startTime);

        const filter = ctx.createBiquadFilter();
        filter.type = 'lowpass';
        filter.frequency.setValueAtTime(1200, startTime);

        gain.gain.setValueAtTime(0.001, startTime);
        gain.gain.linearRampToValueAtTime(0.6, startTime + 0.015);
        gain.gain.exponentialRampToValueAtTime(0.001, startTime + 0.12);

        osc.connect(filter);
        filter.connect(gain);
        gain.connect(ctx.destination);

        osc.start(startTime);
        osc.stop(startTime + 0.13);
      });
    } catch {
      // Audio failsafe
    }
  }

  public playUiClick() {
    if (this.isMuted) return;
    const ctx = this.getContext();
    if (!ctx) return;

    try {
      const now = ctx.currentTime;
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();

      osc.type = 'sine';
      osc.frequency.setValueAtTime(600, now);
      osc.frequency.exponentialRampToValueAtTime(200, now + 0.03);

      gain.gain.setValueAtTime(0.2, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.03);

      osc.connect(gain);
      gain.connect(ctx.destination);

      osc.start(now);
      osc.stop(now + 0.035);
    } catch {
      // Audio failsafe
    }
  }
}

export function triggerHaptic(durationMs: number | number[] = 80) {
  if (typeof window !== 'undefined' && 'navigator' in window && typeof navigator.vibrate === 'function') {
    try {
      navigator.vibrate(durationMs);
    } catch {
      // Failsafe
    }
  }
}

/* =========================================================================
   TIME FORMAT UTILITY
   ========================================================================= */

export function formatClockTime(
  ms: number,
  showTenthsThresholdSec: number = 20
): FormattedTime {
  if (ms <= 0) {
    return {
      display: '0:00.0',
      minutesPart: '0',
      secondsPart: '00',
      tenthsPart: '0',
      isLowTime: true,
      isExpired: true,
    };
  }

  const totalSeconds = Math.floor(ms / 1000);
  const hours = Math.floor(totalSeconds / 3600);
  const minutes = Math.floor((totalSeconds % 3600) / 60);
  const seconds = totalSeconds % 60;
  const tenths = Math.floor((ms % 1000) / 100);

  const isLowTime = totalSeconds <= 10;
  const isSubThreshold = totalSeconds < showTenthsThresholdSec;

  if (hours > 0) {
    const formattedM = minutes.toString().padStart(2, '0');
    const formattedS = seconds.toString().padStart(2, '0');
    return {
      display: `${hours}:${formattedM}:${formattedS}`,
      minutesPart: `${hours}:${formattedM}`,
      secondsPart: formattedS,
      isLowTime: false,
      isExpired: false,
    };
  }

  const formattedM = minutes.toString();
  const formattedS = seconds.toString().padStart(2, '0');

  if (isSubThreshold) {
    return {
      display: `${formattedM}:${formattedS}.${tenths}`,
      minutesPart: formattedM,
      secondsPart: formattedS,
      tenthsPart: tenths.toString(),
      isLowTime,
      isExpired: false,
    };
  }

  return {
    display: `${formattedM}:${formattedS}`,
    minutesPart: formattedM,
    secondsPart: formattedS,
    isLowTime,
    isExpired: false,
  };
}

/* =========================================================================
   CLOCK PANEL COMPONENT (Strict Countdown Only)
   ========================================================================= */

interface ClockPanelProps {
  player: PlayerId;
  playerName: string;
  timeMs: number;
  isActive: boolean;
  gameState: GameState;
  isTimeoutWinner: boolean;
  isTimeoutLoser: boolean;
  rotated: boolean;
  theme: ThemeStyles;
  showTenthsThreshold: number;
  onPanelTap: (player: PlayerId) => void;
}

export const ClockPanel: React.FC<ClockPanelProps> = ({
  player,
  playerName,
  timeMs,
  isActive,
  gameState,
  isTimeoutWinner,
  isTimeoutLoser,
  rotated,
  theme,
  showTenthsThreshold,
  onPanelTap,
}) => {
  const [isJolting, setIsJolting] = useState(false);

  const formatted = useMemo(
    () => formatClockTime(timeMs, showTenthsThreshold),
    [timeMs, showTenthsThreshold]
  );

  const isLowTimeActive = isActive && formatted.isLowTime && gameState === 'running';

  let bgClass = theme.inactiveBg;
  let textClass = theme.inactiveText;

  if (isTimeoutLoser) {
    bgClass = theme.timeoutBg;
    textClass = theme.timeoutText;
  } else if (isTimeoutWinner) {
    bgClass = 'bg-emerald-950';
    textClass = 'text-emerald-100';
  } else if (isLowTimeActive) {
    bgClass = theme.lowTimeActiveBg;
    textClass = theme.lowTimeActiveText;
  } else if (isActive) {
    bgClass = theme.activeBg;
    textClass = theme.activeText;
  }

  const triggerTapAction = () => {
    setIsJolting(true);
    setTimeout(() => setIsJolting(false), 70);
    onPanelTap(player);
  };

  const handleTouch = (e: React.TouchEvent<HTMLDivElement>) => {
    e.preventDefault();
    triggerTapAction();
  };

  const handlePointer = (e: React.PointerEvent<HTMLDivElement>) => {
    if (e.pointerType === 'touch') return;
    if (e.button !== 0 && e.pointerType === 'mouse') return;
    e.preventDefault();
    triggerTapAction();
  };

  return (
    <div
      onTouchStart={handleTouch}
      onPointerDown={handlePointer}
      role="button"
      tabIndex={0}
      aria-label={`${playerName} countdown: ${formatted.display}`}
      className={`relative flex-1 w-full h-full select-none cursor-pointer overflow-hidden transition-none flex items-center justify-center p-4 sm:p-8 ${bgClass} ${textClass} ${
        rotated ? 'rotate-180' : ''
      } ${isJolting ? 'animate-haptic scale-[0.985]' : ''} active:scale-[0.985] touch-manipulation`}
    >
      <div className="relative z-10 flex items-baseline justify-center font-mono font-black tracking-tight select-none pointer-events-none">
        <span className="text-7xl sm:text-9xl md:text-[11rem] lg:text-[13rem] tabular-nums leading-none">
          {formatted.minutesPart}:{formatted.secondsPart}
        </span>
        {formatted.tenthsPart !== undefined && (
          <span className="text-4xl sm:text-6xl md:text-7xl lg:text-8xl tabular-nums opacity-85 ml-1 sm:ml-2">
            .{formatted.tenthsPart}
          </span>
        )}
      </div>
    </div>
  );
};

interface CenterControlsProps {
  gameState: GameState;
  soundEnabled: boolean;
  oppositeFacing: boolean;
  onTogglePlayPause: () => void;
  onResetGame: () => void;
  onOpenSettings: () => void;
  onToggleSound: () => void;
  onToggleOppositeFacing: () => void;
}

export const CenterControls: React.FC<CenterControlsProps> = ({
  gameState,
  soundEnabled,
  oppositeFacing,
  onTogglePlayPause,
  onResetGame,
  onOpenSettings,
  onToggleSound,
  onToggleOppositeFacing,
}) => {
  const [confirmReset, setConfirmReset] = useState(false);
  const [isFullscreen, setIsFullscreen] = useState(false);

  const handleResetClick = (e: React.MouseEvent) => {
    e.stopPropagation();
    soundEngine.playUiClick();
    if (gameState === 'ready') {
      onResetGame();
      return;
    }
    if (confirmReset) {
      onResetGame();
      setConfirmReset(false);
    } else {
      setConfirmReset(true);
      setTimeout(() => {
        setConfirmReset(false);
      }, 3500);
    }
  };

  const handleFullscreenToggle = (e: React.MouseEvent) => {
    e.stopPropagation();
    soundEngine.playUiClick();
    if (!document.fullscreenElement) {
      document.documentElement.requestFullscreen().catch(() => {});
      setIsFullscreen(true);
    } else {
      document.exitFullscreen().catch(() => {});
      setIsFullscreen(false);
    }
  };

  return (
    <div
      onPointerDown={(e) => e.stopPropagation()}
      className="relative z-30 w-full shrink-0 h-14 sm:h-16 bg-zinc-900 border-y border-zinc-800 flex items-center justify-between px-3 sm:px-6 shadow-xl"
    >
      <div className="flex items-center gap-1 sm:gap-2">
        <button
          type="button"
          onClick={(e) => {
            e.stopPropagation();
            soundEngine.playUiClick();
            onOpenSettings();
          }}
          title="Time Controls & Settings"
          aria-label="Settings"
          className="p-2 sm:p-2.5 rounded-lg text-zinc-400 hover:text-white hover:bg-zinc-800 active:scale-95 transition-colors"
        >
          <SettingsIcon className="w-4 h-4 sm:w-5 sm:h-5" />
        </button>

        <button
          type="button"
          onClick={(e) => {
            e.stopPropagation();
            soundEngine.playUiClick();
            onToggleSound();
          }}
          title={soundEnabled ? 'Mute Sounds' : 'Unmute Sounds'}
          aria-label={soundEnabled ? 'Mute Sounds' : 'Unmute Sounds'}
          className={`p-2 sm:p-2.5 rounded-lg active:scale-95 transition-colors ${
            soundEnabled
              ? 'text-zinc-300 hover:text-white hover:bg-zinc-800'
              : 'text-zinc-600 hover:text-zinc-400 hover:bg-zinc-800/60'
          }`}
        >
          {soundEnabled ? (
            <Volume2 className="w-4 h-4 sm:w-5 sm:h-5" />
          ) : (
            <VolumeX className="w-4 h-4 sm:w-5 sm:h-5" />
          )}
        </button>

        <button
          type="button"
          onClick={(e) => {
            e.stopPropagation();
            soundEngine.playUiClick();
            onToggleOppositeFacing();
          }}
          title={oppositeFacing ? 'Opposite Facing (180° enabled)' : 'Same Direction Facing'}
          aria-label="Toggle clock orientation"
          className={`p-2 sm:p-2.5 rounded-lg active:scale-95 transition-colors ${
            oppositeFacing ? 'text-emerald-400 hover:bg-zinc-800' : 'text-zinc-500 hover:text-white hover:bg-zinc-800'
          }`}
        >
          <Smartphone
            className={`w-4 h-4 sm:w-5 sm:h-5 transition-transform ${
              oppositeFacing ? 'rotate-180 text-emerald-400' : ''
            }`}
          />
        </button>
      </div>

      <div className="flex items-center gap-2">
        <button
          type="button"
          onClick={(e) => {
            e.stopPropagation();
            soundEngine.playUiClick();
            onTogglePlayPause();
          }}
          title={gameState === 'running' ? 'Pause Game' : 'Resume Game'}
          aria-label={gameState === 'running' ? 'Pause Game' : 'Resume Game'}
          className={`h-10 sm:h-11 px-4 sm:px-6 rounded-full flex items-center justify-center gap-2 font-mono text-xs sm:text-sm font-semibold tracking-wide uppercase transition-all shadow-md active:scale-95 ${
            gameState === 'running'
              ? 'bg-amber-500 hover:bg-amber-400 text-zinc-950 ring-2 ring-amber-400/40'
              : gameState === 'paused'
              ? 'bg-emerald-500 hover:bg-emerald-400 text-zinc-950 animate-pulse'
              : 'bg-zinc-800 hover:bg-zinc-700 text-zinc-200'
          }`}
        >
          {gameState === 'running' ? (
            <>
              <Pause className="w-4 h-4 fill-current" />
              <span>Pause</span>
            </>
          ) : (
            <>
              <Play className="w-4 h-4 fill-current" />
              <span>{gameState === 'paused' ? 'Resume' : 'Start'}</span>
            </>
          )}
        </button>
      </div>

      <div className="flex items-center gap-1 sm:gap-2">
        <button
          type="button"
          onClick={handleResetClick}
          title={confirmReset ? 'Tap again to confirm reset' : 'Reset Clock'}
          aria-label="Reset Clock"
          className={`p-2 sm:p-2.5 rounded-lg flex items-center gap-1.5 active:scale-95 transition-colors ${
            confirmReset
              ? 'bg-red-500 text-white font-mono text-xs font-bold px-3 py-1.5 animate-pulse'
              : 'text-zinc-400 hover:text-white hover:bg-zinc-800'
          }`}
        >
          {confirmReset ? (
            <>
              <Check className="w-4 h-4" />
              <span>Reset?</span>
            </>
          ) : (
            <RotateCcw className="w-4 h-4 sm:w-5 sm:h-5" />
          )}
        </button>

        <button
          type="button"
          onClick={handleFullscreenToggle}
          title="Toggle Fullscreen"
          aria-label="Toggle Fullscreen"
          className="p-2 sm:p-2.5 rounded-lg text-zinc-400 hover:text-white hover:bg-zinc-800 active:scale-95 transition-colors hidden xs:flex"
        >
          {isFullscreen ? (
            <Minimize2 className="w-4 h-4 sm:w-5 sm:h-5" />
          ) : (
            <Maximize2 className="w-4 h-4 sm:w-5 sm:h-5" />
          )}
        </button>
      </div>
    </div>
  );
};

/* =========================================================================
   MAIN APP COMPONENT
   ========================================================================= */

export default function App() {
  const [settings, setSettings] = useState<ClockSettings>(DEFAULT_SETTINGS);
  const [gameState, setGameState] = useState<GameState>('ready');
  const [activePlayer, setActivePlayer] = useState<PlayerId | null>(null);

  const [p1TimeMs, setP1TimeMs] = useState<number>(
    (DEFAULT_SETTINGS.p1BaseMinutes * 60 + DEFAULT_SETTINGS.p1BaseSeconds) * 1000
  );
  const [p2TimeMs, setP2TimeMs] = useState<number>(
    (DEFAULT_SETTINGS.p2BaseMinutes * 60 + DEFAULT_SETTINGS.p2BaseSeconds) * 1000
  );

  const [p1Moves, setP1Moves] = useState<number>(0);
  const [p2Moves, setP2Moves] = useState<number>(0);
  const [timeoutWinner, setTimeoutWinner] = useState<PlayerId | null>(null);
  const [isSettingsOpen, setIsSettingsOpen] = useState<boolean>(false);

  const p1TimeRef = useRef<number>(p1TimeMs);
  const p2TimeRef = useRef<number>(p2TimeMs);
  const activePlayerRef = useRef<PlayerId | null>(activePlayer);
  const gameStateRef = useRef<GameState>(gameState);
  const settingsRef = useRef<ClockSettings>(settings);

  p1TimeRef.current = p1TimeMs;
  p2TimeRef.current = p2TimeMs;
  activePlayerRef.current = activePlayer;
  gameStateRef.current = gameState;
  settingsRef.current = settings;

  useEffect(() => {
    const handleFirstTouch = () => {
      soundEngine.init();
      window.removeEventListener('pointerdown', handleFirstTouch);
      window.removeEventListener('keydown', handleFirstTouch);
    };
    window.addEventListener('pointerdown', handleFirstTouch, { once: true });
    window.addEventListener('keydown', handleFirstTouch, { once: true });

    return () => {
      window.removeEventListener('pointerdown', handleFirstTouch);
      window.removeEventListener('keydown', handleFirstTouch);
    };
  }, []);

  useEffect(() => {
    soundEngine.setMuted(!settings.soundEnabled);
  }, [settings.soundEnabled]);

  // High-precision clock tick loop
  useEffect(() => {
    if (gameState !== 'running' || !activePlayer) return;

    let lastTimestamp = performance.now();
    let frameId: number;

    const tick = (now: DOMHighResTimeStamp) => {
      const delta = now - lastTimestamp;
      lastTimestamp = now;

      const currentActive = activePlayerRef.current;
      if (!currentActive || gameStateRef.current !== 'running') return;

      const currentSettings = settingsRef.current;
      const isPlayer1 = currentActive === 1;
      const currentRemaining = isPlayer1 ? p1TimeRef.current : p2TimeRef.current;
      const nextRemaining = Math.max(0, currentRemaining - delta);

      const prevSec = Math.ceil(currentRemaining / 1000);
      const nextSec = Math.ceil(nextRemaining / 1000);

      // Low-time warning audio (exact once per second tick <= 10s)
      if (
        currentSettings.lowTimeWarningEnabled &&
        currentSettings.soundEnabled &&
        nextRemaining > 0 &&
        nextSec <= 10 &&
        nextSec < prevSec
      ) {
        soundEngine.playLowTimeWarning(nextSec);
        if (currentSettings.hapticsEnabled) {
          triggerHaptic(80);
        }
      }

      if (isPlayer1) {
        p1TimeRef.current = nextRemaining;
        setP1TimeMs(nextRemaining);
      } else {
        p2TimeRef.current = nextRemaining;
        setP2TimeMs(nextRemaining);
      }

      // Timeout check
      if (nextRemaining <= 0) {
        if (currentSettings.soundEnabled) {
          soundEngine.playTimeoutAlarm();
        }
        if (currentSettings.hapticsEnabled) {
          triggerHaptic([120, 60, 120, 60, 250]);
        }
        setGameState('timeout');
        setTimeoutWinner(isPlayer1 ? 2 : 1);
        return;
      }

      frameId = requestAnimationFrame(tick);
    };

    frameId = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(frameId);
  }, [gameState, activePlayer]);

  // Instantaneous turn-switching tap handler
  const handlePlayerTap = useCallback(
    (player: PlayerId) => {
      soundEngine.init();
      const currentSettings = settingsRef.current;
      if (currentSettings.hapticsEnabled) {
        triggerHaptic(80);
      }
      if (currentSettings.soundEnabled) {
        soundEngine.playMechanicalClick();
      }

      const currentGameState = gameStateRef.current;
      const currentActive = activePlayerRef.current;

      if (currentGameState === 'timeout') {
        handleResetGame();
        return;
      }

      if (currentGameState === 'paused') {
        gameStateRef.current = 'running';
        setGameState('running');
        return;
      }

      if (currentGameState === 'ready') {
        const nextActive: PlayerId = player === 1 ? 2 : 1;
        activePlayerRef.current = nextActive;
        gameStateRef.current = 'running';
        setActivePlayer(nextActive);
        setGameState('running');

        if (player === 1) {
          setP1Moves(1);
          if (currentSettings.p1IncrementSeconds > 0) {
            const incMs = currentSettings.p1IncrementSeconds * 1000;
            p1TimeRef.current += incMs;
            setP1TimeMs((t) => t + incMs);
          }
        } else {
          setP2Moves(1);
          if (currentSettings.p2IncrementSeconds > 0) {
            const incMs = currentSettings.p2IncrementSeconds * 1000;
            p2TimeRef.current += incMs;
            setP2TimeMs((t) => t + incMs);
          }
        }
        return;
      }

      if (currentGameState === 'running') {
        const whoMoved: PlayerId = currentActive || player;
        const nextActive: PlayerId = whoMoved === 1 ? 2 : 1;

        if (whoMoved === 1) {
          setP1Moves((m) => m + 1);
          if (currentSettings.p1IncrementSeconds > 0) {
            const incMs = currentSettings.p1IncrementSeconds * 1000;
            p1TimeRef.current += incMs;
            setP1TimeMs((t) => t + incMs);
          }
        } else {
          setP2Moves((m) => m + 1);
          if (currentSettings.p2IncrementSeconds > 0) {
            const incMs = currentSettings.p2IncrementSeconds * 1000;
            p2TimeRef.current += incMs;
            setP2TimeMs((t) => t + incMs);
          }
        }

        activePlayerRef.current = nextActive;
        setActivePlayer(nextActive);
      }
    },
    []
  );

  const handleTogglePlayPause = () => {
    if (gameState === 'ready') {
      setActivePlayer(1);
      setGameState('running');
    } else if (gameState === 'running') {
      setGameState('paused');
    } else if (gameState === 'paused') {
      setGameState('running');
    } else if (gameState === 'timeout') {
      handleResetGame();
    }
  };

  const handleResetGame = () => {
    setGameState('ready');
    setActivePlayer(null);
    setTimeoutWinner(null);
    setP1Moves(0);
    setP2Moves(0);

    const p1Ms = (settings.p1BaseMinutes * 60 + settings.p1BaseSeconds) * 1000;
    const p2Ms = (settings.p2BaseMinutes * 60 + settings.p2BaseSeconds) * 1000;
    setP1TimeMs(p1Ms);
    setP2TimeMs(p2Ms);
    p1TimeRef.current = p1Ms;
    p2TimeRef.current = p2Ms;
  };

  const handleSaveSettings = (newSettings: ClockSettings) => {
    setSettings(newSettings);
    setGameState('ready');
    setActivePlayer(null);
    setTimeoutWinner(null);
    setP1Moves(0);
    setP2Moves(0);

    const p1Ms = (newSettings.p1BaseMinutes * 60 + newSettings.p1BaseSeconds) * 1000;
    const p2Ms = (newSettings.p2BaseMinutes * 60 + newSettings.p2BaseSeconds) * 1000;
    setP1TimeMs(p1Ms);
    setP2TimeMs(p2Ms);
    p1TimeRef.current = p1Ms;
    p2TimeRef.current = p2Ms;
  };

  const handleAdjustTime = (player: PlayerId, deltaSeconds: number) => {
    const deltaMs = deltaSeconds * 1000;
    if (player === 1) {
      setP1TimeMs((t) => {
        const next = Math.max(0, t + deltaMs);
        p1TimeRef.current = next;
        return next;
      });
    } else {
      setP2TimeMs((t) => {
        const next = Math.max(0, t + deltaMs);
        p2TimeRef.current = next;
        return next;
      });
    }
  };

  // Keyboard navigation for desktop
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (
        isSettingsOpen ||
        document.activeElement?.tagName === 'INPUT' ||
        document.activeElement?.tagName === 'TEXTAREA'
      ) {
        return;
      }

      if (e.code === 'Space') {
        e.preventDefault();
        if (gameState === 'running' && activePlayer) {
          handlePlayerTap(activePlayer);
        } else {
          handleTogglePlayPause();
        }
      } else if (e.code === 'KeyP') {
        e.preventDefault();
        handleTogglePlayPause();
      } else if (e.code === 'KeyR') {
        e.preventDefault();
        handleResetGame();
      } else if (e.code === 'ArrowLeft' || e.code === 'KeyZ' || e.code === 'Digit1') {
        e.preventDefault();
        handlePlayerTap(1);
      } else if (e.code === 'ArrowRight' || e.code === 'Slash' || e.code === 'KeyM' || e.code === 'Digit2') {
        e.preventDefault();
        handlePlayerTap(2);
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [gameState, activePlayer, isSettingsOpen, handlePlayerTap]);

  const activeTheme = THEMES[settings.theme] || THEMES.tournament;

  return (
    <main className="w-screen h-screen overflow-hidden flex flex-col bg-zinc-950 text-white select-none touch-none">
      <ClockPanel
        player={1}
        playerName="Player 1"
        timeMs={p1TimeMs}
        isActive={activePlayer === 1 && gameState === 'running'}
        gameState={gameState}
        isTimeoutWinner={timeoutWinner === 1}
        isTimeoutLoser={timeoutWinner === 2}
        rotated={settings.oppositeFacing}
        theme={activeTheme}
        showTenthsThreshold={settings.showTenthsThresholdSeconds}
        onPanelTap={handlePlayerTap}
      />

      <CenterControls
        gameState={gameState}
        soundEnabled={settings.soundEnabled}
        oppositeFacing={settings.oppositeFacing}
        onTogglePlayPause={handleTogglePlayPause}
        onResetGame={handleResetGame}
        onOpenSettings={() => setIsSettingsOpen(true)}
        onToggleSound={() =>
          setSettings((prev) => ({ ...prev, soundEnabled: !prev.soundEnabled }))
        }
        onToggleOppositeFacing={() =>
          setSettings((prev) => ({
            ...prev,
            oppositeFacing: !prev.oppositeFacing,
          }))
        }
      />

      <ClockPanel
        player={2}
        playerName="Player 2"
        timeMs={p2TimeMs}
        isActive={activePlayer === 2 && gameState === 'running'}
        gameState={gameState}
        isTimeoutWinner={timeoutWinner === 2}
        isTimeoutLoser={timeoutWinner === 1}
        rotated={false}
        theme={activeTheme}
        showTenthsThreshold={settings.showTenthsThresholdSeconds}
        onPanelTap={handlePlayerTap}
      />

      <SettingsModal
        isOpen={isSettingsOpen}
        onClose={() => setIsSettingsOpen(false)}
        settings={settings}
        onSaveSettings={handleSaveSettings}
        onAdjustTime={handleAdjustTime}
        p1TimeMs={p1TimeMs}
        p2TimeMs={p2TimeMs}
      />
    </main>
  );
}
